DATABASE SCHEMA (Firestore Collections)
1. Users Collection
javascript
{
  uid: string,                  // Firebase Auth UID
  email: string,
  role: 'student' | 'enterprise' | 'admin',
  displayName: string,
  phone: string,
  preferredLanguage: string,    // For UI localization
  createdAt: timestamp,
  updatedAt: timestamp,
  isActive: boolean
}
2. Students Collection
javascript
{
  uid: string,                  // References Users.uid
  studentId: string,
  major: string,
  grade: string,               // e.g., "Freshman", "Sophomore", etc.
  year: number,                // 1, 2, 3, 4
  idCardPhoto: string,         // Firebase Storage URL
  isVerified: boolean,
  verifiedAt: timestamp,
  verifiedBy: string,          // Admin uid
  skills: string[],            // Extracted from resume
  availability: string,        // e.g., "Weekends", "Weekdays evening"
  createdAt: timestamp,
  updatedAt: timestamp
}
3. Enterprises Collection
javascript
{
  uid: string,                  // References Users.uid
  companyName: string,
  businessLicense: string,     // Storage URL
  storePhotos: string[],       // Storage URLs
  description: string,
  contactPerson: string,
  contactPhone: string,
  address: string,
  website: string,
  isApproved: boolean,
  approvedAt: timestamp,
  approvedBy: string,          // Admin uid
  isBanned: boolean,
  banReason: string,
  createdAt: timestamp,
  updatedAt: timestamp
}
4. Resumes Collection
javascript
{
  id: string,                   // Auto-generated
  studentUid: string,          // References Students.uid
  major: string,
  grade: string,
  skills: string[],
  availableHours: string,      // e.g., "20 hours/week", "Weekends only"
  experience: string,          // Text description of previous experience
  certificates: string[],      // Storage URLs
  isDefault: boolean,
  createdAt: timestamp,
  updatedAt: timestamp
}
5. Jobs Collection
javascript
{
  id: string,                   // Auto-generated
  enterpriseUid: string,       // References Enterprises.uid
  title: string,
  responsibilities: string,
  salary: string,              // e.g., "$15/hour" or "Monthly salary"
  workingHours: string,        // e.g., "9 AM - 5 PM"
  quota: number,               // Number of positions
  location: string,
  skillRequirements: string[],
  duration: string,            // e.g., "3 months", "Summer internship"
  status: 'pending' | 'active' | 'paused' | 'expired' | 'rejected',
  views: number,
  applicationsCount: number,
  postedAt: timestamp,
  expiresAt: timestamp,        // Auto-expiry after 30 days
  updatedAt: timestamp,
  auditNotes: string,          // Admin notes
  auditedBy: string,          // Admin uid
  auditAt: timestamp
}
6. Applications Collection
javascript
{
  id: string,                   // Auto-generated
  jobId: string,               // References Jobs.id
  studentUid: string,          // References Students.uid
  resumeId: string,            // References Resumes.id
  status: 'pending' | 'reviewing' | 'interview' | 'hired' | 'rejected' | 'completed',
  interviewSchedule: {
    date: timestamp,
    location: string,
    notes: string,
    sentAt: timestamp
  },
  appliedAt: timestamp,
  updatedAt: timestamp,
  notes: string,              // Additional notes from enterprise
  hiredAt: timestamp,
  completedAt: timestamp
}
7. Announcements Collection
javascript
{
  id: string,                   // Auto-generated
  title: string,
  content: string,
  type: 'announcement' | 'article',
  bannerImage: string,         // Storage URL (optional)
  isActive: boolean,
  priority: number,            // For ordering
  createdAt: timestamp,
  createdBy: string,           // Admin uid
  updatedAt: timestamp
}
8. Banners Collection
javascript
{
  id: string,                   // Auto-generated
  imageUrl: string,            // Storage URL
  link: string,                // Where the banner redirects
  title: string,
  order: number,               // Display order
  isActive: boolean,
  createdAt: timestamp,
  updatedAt: timestamp
}
9. Notifications Collection
javascript
{
  id: string,                   // Auto-generated
  userId: string,              // Recipient uid
  type: 'application_status' | 'interview_scheduled' | 'job_approved' | 'system',
  title: string,
  message: string,
  isRead: boolean,
  data: object,                // Additional data (e.g., applicationId)
  createdAt: timestamp
}
BACKEND API STRUCTURE (FastAPI)
API Endpoints Design
Authentication Endpoints
text
POST   /api/v1/auth/register          # Register new user
POST   /api/v1/auth/login             # Login user
POST   /api/v1/auth/logout            # Logout user
POST   /api/v1/auth/refresh-token     # Refresh access token
POST   /api/v1/auth/forgot-password   # Send password reset email
POST   /api/v1/auth/reset-password    # Reset password with token
GET    /api/v1/auth/verify-email      # Verify email with token
GET    /api/v1/auth/me                # Get current user info
Student Endpoints
text
GET    /api/v1/students/profile               # Get student profile
PUT    /api/v1/students/profile               # Update student profile
POST   /api/v1/students/verify-id             # Upload student ID for verification
GET    /api/v1/students/verification-status   # Get verification status
GET    /api/v1/students/applications          # Get all applications of student
GET    /api/v1/students/applications/:id      # Get specific application
Resume Endpoints
text
GET    /api/v1/resumes                 # Get all resumes of current student
GET    /api/v1/resumes/:id             # Get specific resume
POST   /api/v1/resumes                 # Create new resume
PUT    /api/v1/resumes/:id             # Update resume
DELETE /api/v1/resumes/:id             # Delete resume
POST   /api/v1/resumes/:id/certificate # Upload certificate
DELETE /api/v1/resumes/:id/certificate/:certId # Remove certificate
PUT    /api/v1/resumes/:id/set-default # Set resume as default
Job Endpoints (Student view)
text
GET    /api/v1/jobs                    # List jobs with filters
GET    /api/v1/jobs/:id                # Get job details
GET    /api/v1/jobs/:id/check-application # Check if student applied
POST   /api/v1/jobs/:id/apply         # Apply to job
GET    /api/v1/jobs/recommended       # Get recommended jobs
GET    /api/v1/jobs/saved             # Get saved jobs
POST   /api/v1/jobs/:id/save          # Save job
DELETE /api/v1/jobs/:id/save          # Unsave job
Enterprise Job Endpoints
text
GET    /api/v1/enterprise/jobs              # Get enterprise's jobs
POST   /api/v1/enterprise/jobs              # Create new job posting
PUT    /api/v1/enterprise/jobs/:id          # Update job
DELETE /api/v1/enterprise/jobs/:id          # Delete job
PATCH  /api/v1/enterprise/jobs/:id/status   # Update job status
GET    /api/v1/enterprise/jobs/:id/applications # Get applications for job
Application Endpoints (Enterprise)
text
GET    /api/v1/enterprise/applications              # Get all applications
GET    /api/v1/enterprise/applications/:id          # Get specific application
PUT    /api/v1/enterprise/applications/:id/status   # Update application status
POST   /api/v1/enterprise/applications/:id/interview # Schedule interview
GET    /api/v1/enterprise/applications/:id/resume   # View student resume
Admin Endpoints
text
# User Management
GET    /api/v1/admin/users                   # List all users
GET    /api/v1/admin/users/:uid              # Get user details
PUT    /api/v1/admin/users/:uid/role         # Update user role
POST   /api/v1/admin/users/:uid/verify-student # Verify student
POST   /api/v1/admin/users/:uid/approve-enterprise # Approve enterprise
POST   /api/v1/admin/users/:uid/ban          # Ban user
POST   /api/v1/admin/users/:uid/unban        # Unban user

# Job Management
GET    /api/v1/admin/jobs/pending            # Get pending jobs
GET    /api/v1/admin/jobs                    # Get all jobs
POST   /api/v1/admin/jobs/:id/approve        # Approve job
POST   /api/v1/admin/jobs/:id/reject         # Reject job
POST   /api/v1/admin/jobs/:id/remove         # Remove job from platform

# Platform Configuration
GET    /api/v1/admin/banners                 # Get all banners
POST   /api/v1/admin/banners                 # Create banner
PUT    /api/v1/admin/banners/:id             # Update banner
DELETE /api/v1/admin/banners/:id             # Delete banner

GET    /api/v1/admin/announcements           # Get all announcements
POST   /api/v1/admin/announcements           # Create announcement
PUT    /api/v1/admin/announcements/:id       # Update announcement
DELETE /api/v1/admin/announcements/:id       # Delete announcement
Notification Endpoints
text
GET    /api/v1/notifications                 # Get user notifications
PATCH  /api/v1/notifications/:id/read       # Mark notification as read
POST   /api/v1/notifications/mark-all-read  # Mark all as read
DELETE /api/v1/notifications/:id            # Delete notification
